<script>
const NEWSFY_API="https://newsfy.ricardorodrigues0671.worker.dev";
let todasNoticias=[];

function escapar(v){return String(v??"").replace(/&/g,"&amp;").replace(/</g,"&lt;").replace(/>/g,"&gt;").replace(/"/g,"&quot;").replace(/'/g,"&#039;");}
function textoCampo(v){if(v==null)return "";if(typeof v==="object"){if("value" in v)return String(v.value||"");return "";}return String(v);}
function extrairImagem(v){
if(!v)return "";
if(typeof v==="string"){let t=v.trim(),m=t.match(/\]\((https?:\/\/[^)]+)\)/);if(m&&m[1])return m[1];if(/^https?:\/\//.test(t))return t;return "";}
if(Array.isArray(v)&&v.length){let a=v[0];if(a?.url)return a.url;if(a?.thumbnails?.large?.url)return a.thumbnails.large.url;}
if(typeof v==="object"&&v.url)return v.url; return "";
}
function imagemDaNoticia(n){return extrairImagem(n["Imagem capa"])||extrairImagem(n["Imagem"]);}
function estaPublicada(n){let s=n["Status"];if(!s)return false;if(typeof s==="object"&&s.value)return String(s.value).trim().toLowerCase()==="publicado";return String(s).trim().toLowerCase()==="publicado";}
function ehDestaque(n){let v=n["Destaque"];return v===true||v===1||String(v).toLowerCase()==="true";}
function categoriaDaNoticia(n){let c=n["Categoria"];if(c&&typeof c==="object"&&c.value)return String(c.value);return textoCampo(c)||"Geral";}
function dataDaNoticia(n){let v=n["Data publicação"];if(!v)return "";try{return new Date(v).toLocaleDateString("pt-BR",{day:"2-digit",month:"2-digit",year:"numeric"});}catch{return "";}}
function urlDaNoticia(n){return "noticia.html?id="+encodeURIComponent(n.id||n["ID"]);}

function criarCard(n){
let t=textoCampo(n["Titulo"])||"Sem título",r=textoCampo(n["Resumo"]),c=categoriaDaNoticia(n),d=dataDaNoticia(n),i=imagemDaNoticia(n);
return `<article class="news-card"><a href="${urlDaNoticia(n)}" class="news-image">${i?`<img src="${escapar(i)}" alt="${escapar(t)}" loading="lazy">`:`<div class="image-placeholder">NEWSFY</div>`}</a><div class="news-card-body"><span class="category">${escapar(c)}</span><h3><a href="${urlDaNoticia(n)}">${escapar(t)}</a></h3>${r?`<p>${escapar(r)}</p>`:""}<time>${escapar(d)}</time></div></article>`;
}
function criarListaItem(n,num){let t=textoCampo(n["Titulo"])||"Sem título";return `<a href="${urlDaNoticia(n)}" class="trending-item"><strong>${num}</strong><span>${escapar(t)}</span></a>`;}

function renderizarNoticias(){
const hero=document.getElementById("heroArticle"),featured=document.getElementById("featuredGrid"),latest=document.getElementById("latestList"),trending=document.getElementById("trendingList");
if(!todasNoticias.length){hero.innerHTML="<p>Nenhuma notícia publicada encontrada.</p>";featured.innerHTML="<p>Nenhuma notícia encontrada.</p>";latest.innerHTML="";trending.innerHTML="";return;}
const destaque=todasNoticias.find(ehDestaque)||todasNoticias[0],t=textoCampo(destaque["Titulo"])||"Sem título",r=textoCampo(destaque["Resumo"]),i=imagemDaNoticia(destaque);
hero.innerHTML=`<article class="hero-card"><a href="${urlDaNoticia(destaque)}" class="hero-image">${i?`<img src="${escapar(i)}" alt="${escapar(t)}">`:`<div class="image-placeholder">NEWSFY</div>`}</a><div class="hero-content"><span class="category">${escapar(categoriaDaNoticia(destaque))}</span><h2><a href="${urlDaNoticia(destaque)}">${escapar(t)}</a></h2>${r?`<p>${escapar(r)}</p>`:""}<time>${escapar(dataDaNoticia(destaque))}</time></div></article>`;
let ds=todasNoticias.filter(n=>n.id!==destaque.id&&ehDestaque(n)).slice(0,6);if(!ds.length)ds=todasNoticias.filter(n=>n.id!==destaque.id).slice(0,6);
featured.innerHTML=ds.length?ds.map(criarCard).join(""):"<p>Nenhuma notícia em destaque.</p>";
let recentes=todasNoticias.filter(n=>n.id!==destaque.id).slice(0,10);
latest.innerHTML=recentes.length?recentes.map(criarCard).join(""):"<p>Nenhuma notícia recente.</p>";
trending.innerHTML=todasNoticias.slice(0,5).map((n,x)=>criarListaItem(n,x+1)).join("");
}

async function carregarNoticias(){
try{
const r=await fetch(`${NEWSFY_API}/noticias?size=100`,{headers:{"Accept":"application/json"},cache:"no-store"});
if(!r.ok)throw new Error("HTTP "+r.status);
const dados=await r.json(),resultados=Array.isArray(dados)?dados:(Array.isArray(dados.results)?dados.results:[]);
todasNoticias=resultados.filter(estaPublicada).sort((a,b)=>new Date(b["Data publicação"]||b["Criado em"]||0)-new Date(a["Data publicação"]||a["Criado em"]||0));
renderizarNoticias();
}catch(e){
console.error("NEWSFY:",e);
document.getElementById("heroArticle").innerHTML='<div class="error-box"><strong>Não foi possível carregar as notícias.</strong><p>Verifique a conexão com o servidor.</p></div>';
document.getElementById("featuredGrid").innerHTML="";document.getElementById("latestList").innerHTML="";document.getElementById("trendingList").innerHTML="";
}}
function pesquisarNoticias(termo){
let busca=termo.trim().toLowerCase();if(!busca){renderizarNoticias();return;}
let original=todasNoticias;todasNoticias=original.filter(n=>[n["Titulo"],n["Resumo"],n["Tags"],categoriaDaNoticia(n)].map(textoCampo).join(" ").toLowerCase().includes(busca));renderizarNoticias();todasNoticias=original;
}

document.querySelectorAll(".category-btn").forEach(btn=>btn.addEventListener("click",()=>{
document.querySelectorAll(".category-btn").forEach(b=>b.classList.remove("active"));btn.classList.add("active");
let c=btn.dataset.category,original=todasNoticias;if(c!=="Todas")todasNoticias=original.filter(n=>categoriaDaNoticia(n).toLowerCase()===c.toLowerCase());renderizarNoticias();todasNoticias=original;
}));
document.getElementById("searchBtn").addEventListener("click",()=>{document.getElementById("searchBox").classList.toggle("hidden");document.getElementById("searchInput").focus();});
document.getElementById("searchInput").addEventListener("input",e=>pesquisarNoticias(e.target.value));
document.getElementById("menuBtn").addEventListener("click",()=>document.getElementById("mainNav").classList.toggle("open"));
document.getElementById("themeBtn").addEventListener("click",()=>{document.body.classList.toggle("light-theme");localStorage.setItem("newsfy-theme",document.body.classList.contains("light-theme")?"light":"dark");});
if(localStorage.getItem("newsfy-theme")==="light")document.body.classList.add("light-theme");
document.getElementById("newsletterForm").addEventListener("submit",e=>{e.preventDefault();document.getElementById("newsletterMsg").textContent="Cadastro recebido com sucesso!";e.target.reset();});
carregarNoticias();
